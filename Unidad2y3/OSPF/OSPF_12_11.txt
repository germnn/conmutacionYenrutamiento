
!--------------- Router 1
enable
conf t
hostname R1
no ip domain-lookup
banner motd #SOlo personal autorizado#
ipv6 unicast-routing

!===Enrutamiendo Dinamico v6=== 
ipv6 router ospf 2
router-id 1.1.1.1
passive-interface g0/0
exit

!===Interfaces===

interface s0/0/0
description Red WAN A r2
ip address 192.168.1.1 255.255.255.0
ipv6 add 192:1::1/64
ipv6 add FE80::1 link-local
ipv6 ospf 2 area 0
clock rate 128000
no shutdown 
exit

int g0/0
description RED LAN r1
ip address 11.255.255.254 255.0.0.0
ipv6 add 11::254/64
ipv6 add FE80::1 link-local
ipv6 ospf 2 area 0
ipv6 nd managed-config-flag
ipv6 dhcp server DHCPV6-LAN1
no shutdown
exit

int lo1
description loopback r1
ip address 200.33.16.1 255.255.255.0
ipv6 add 16::1/64
ipv6 add FE80::1 link-local
ipv6 ospf 2 area 0
exit

!===Enrutamiendo Dinamico v4=== 
router ospf 2
router-id 1.1.1.1
passive-interface g0/0
network 192.168.1.0 0.0.0.255 area 0
network 11.0.0.0 0.255.255.255 area 0
network 200.33.16.0 0.0.0.255 area 0
no auto-summary
exit


!===DHCP===
ip dhcp excluded-address 11.0.0.1 11.0.0.2
ip dhcp excluded-address 11.255.255.254 
ip dhcp pool DHCP-POOL-LAN1
network 11.0.0.0 255.0.0.0
default-router 11.255.255.254
exit

ipv6 dhcp pool DHCPV6-LAN1
address prefix 11::/64
dns-server 11::254
domain-name cisco.com
exit
