Router1
enable
configure terminal
hostname R1
no ip domain-lookup

! Interfaz hacia LAN A
interface GigabitEthernet0/0/0
 ip address 192.168.10.1 255.255.255.0
 no shutdown
 exit

! Interfaz de interconexión (WAN) hacia R2
interface GigabitEthernet0/0/1
 ip address 10.0.0.1 255.255.255.252
 no shutdown
 exit

! Ruta estática con siguiente salto (hacia LAN B pasando por la IP de R2)
ip route 192.168.20.0 255.255.255.0 10.0.0.2

---------------------
Router2
enable
configure terminal
hostname R2
no ip domain-lookup


! Interfaz hacia LAN B
interface GigabitEthernet0/0/0
 ip address 192.168.20.1 255.255.255.0
 no shutdown
 exit

! Interfaz de interconexión (WAN) hacia R1
interface GigabitEthernet0/0/1
 ip address 10.0.0.2 255.255.255.252
 no shutdown
 exit

! Ruta estática con siguiente salto (hacia LAN A pasando por la IP de R1)
ip route 192.168.10.0 255.255.255.0 10.0.0.1

---------------------
Switch1 (LAN A)
enable
configure terminal
hostname SW1

! Interfaz de Administración VLAN 1
interface vlan 1
 ip address 192.168.10.2 255.255.255.0
 no shutdown
 exit

! Gateway para poder ser administrado desde fuera de la LAN A
ip default-gateway 192.168.10.1


-----------------------
Switch2 (LAN B)

configure terminal

! Interfaz de Administración VLAN 1

 ip address 192.168.20.2 255.255.255.0
 no shutdown
 exit

! Gateway para poder ser administrado desde fuera de la LAN B
ip default-gateway 192.168.20.1
