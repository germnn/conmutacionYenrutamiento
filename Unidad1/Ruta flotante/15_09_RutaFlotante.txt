Configuracion PC-Router3
IPV4
address -> 192.168.2.2 
mascara -> 255.255.255.0
gateway -> 192.168.2.1
---
IPV6
address -> 2001:DB8:CAFE:2::2/64
gateway -> FE80::3

-- 	Router 3

enable
conf t
hostname R3
ipv6 unicast-routing
no ip domain-lookup

int g0/0
Description Enlace_Switch
ip address 192.168.2.1 255.255.255.0
ipv6 address 2001:db8:cafe:2::1/64
ipv6 address fe80::3 link-local
no shutdown
exit

int s0/1/0
description Enlace_R3_R1
ip address 10.10.10.2 255.255.255.0
ipv6 address 2001:db8:feed:10::2/64
clock rate 64000
no shutdown
exit

int s0/1/1
description EnlaceFlotante_R3_R2
ip address 192.168.1.1 255.255.255.0
ipv6 address 2001:db8:cafe:1::1/64
clock rate 64000
no shutdown
exit

ip route 0.0.0.0 0.0.0.0 s0/1/0
ip route 0.0.0.0 0.0.0.0 s0/1/1 5

ipv6 route ::/0 s0/1/0
ipv6 route ::/0 s0/1/1 5
